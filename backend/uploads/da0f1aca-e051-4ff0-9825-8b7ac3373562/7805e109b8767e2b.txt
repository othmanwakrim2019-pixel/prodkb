        static List<string> AnalyzeDependenciesAndOrder(IOrganizationService service, List<string> entityNames)
        {
            Console.WriteLine("\nAnalyse en cours...\n");

            var existingEntities = new Dictionary<string, EntityMetadata>();
            var nonExistentEntities = new List<string>();

            foreach (var entityName in entityNames)
            {
                Console.Write($"Vérification de '{entityName}'... ");
                try
                {
                    var request = new RetrieveEntityRequest
                    {
                        LogicalName = entityName,
                        EntityFilters = EntityFilters.Entity | EntityFilters.Relationships
                    };
                    var response = (RetrieveEntityResponse)service.Execute(request);
                    existingEntities[entityName] = response.EntityMetadata;
                    Console.WriteLine("? Trouvée");
                }
                catch
                {
                    Console.WriteLine("? N'existe pas");
                    nonExistentEntities.Add(entityName);
                }
            }

            if (nonExistentEntities.Count > 0)
            {
                Console.WriteLine($"\n? {nonExistentEntities.Count} entité(s) non trouvée(s) - elles seront ignorées");
            }

            if (existingEntities.Count == 0)
            {
                return new List<string>();
            }

            Console.WriteLine("\nAnalyse des dépendances...");
            var dependencyGraph = BuildDependencyGraph(service, existingEntities);

            Console.WriteLine("\nDépendances détectées:");
            bool hasDependencies = false;
            foreach (var kvp in dependencyGraph)
            {
                if (kvp.Value.Count > 0)
                {
                    hasDependencies = true;
                    Console.WriteLine($"  '{kvp.Key}' dépend de:");
                    foreach (var dep in kvp.Value)
                    {
                        Console.WriteLine($"    ? {dep}");
                    }
                }
            }

            if (!hasDependencies)
            {
                Console.WriteLine("  Aucune dépendance entre les entités sélectionnées");
            }

            var deletionOrder = TopologicalSort(dependencyGraph);

            return deletionOrder;
        }

        static Dictionary<string, HashSet<string>> BuildDependencyGraph(
            IOrganizationService service,
            Dictionary<string, EntityMetadata> entities)
        {
            var graph = new Dictionary<string, HashSet<string>>();

            foreach (var entityName in entities.Keys)
            {
                graph[entityName] = new HashSet<string>();
            }

            foreach (var kvp in entities)
            {
                var entityName = kvp.Key;
                var metadata = kvp.Value;

                if (metadata.OneToManyRelationships != null)
                {
                    foreach (var relation in metadata.OneToManyRelationships)
                    {
                        var referencingEntity = relation.ReferencingEntity;

                        if (entities.ContainsKey(referencingEntity) && referencingEntity != entityName)
                        {
                            graph[entityName].Add(referencingEntity);
                        }
                    }
                }

                if (metadata.ManyToOneRelationships != null)
                {
                    foreach (var relation in metadata.ManyToOneRelationships)
                    {
                        var referencedEntity = relation.ReferencedEntity;

                        if (entities.ContainsKey(referencedEntity) && referencedEntity != entityName)
                        {
                            graph[referencedEntity].Add(entityName);
                        }
                    }
                }

                if (metadata.ManyToManyRelationships != null)
                {
                    foreach (var relation in metadata.ManyToManyRelationships)
                    {
                        var entity1 = relation.Entity1LogicalName;
                        var entity2 = relation.Entity2LogicalName;

                        if (entity1 == entityName && entities.ContainsKey(entity2))
                        {
                            graph[entity2].Add(entityName);
                        }
                    }
                }
            }

            return graph;
        }

        static List<string> TopologicalSort(Dictionary<string, HashSet<string>> graph)
        {
            var result = new List<string>();
            var visited = new HashSet<string>();
            var visiting = new HashSet<string>();

            bool HasCycle(string node)
            {
                if (visiting.Contains(node))
                    return true;

                if (visited.Contains(node))
                    return false;

                visiting.Add(node);

                foreach (var dependency in graph[node])
                {
                    if (HasCycle(dependency))
                        return true;
                }

                visiting.Remove(node);
                visited.Add(node);
                result.Add(node);

                return false;
            }

            foreach (var node in graph.Keys)
            {
                if (!visited.Contains(node))
                {
                    if (HasCycle(node))
                    {
                        Console.WriteLine("\n? ATTENTION: Dépendances circulaires détectées!");
                        Console.WriteLine("Tentative de suppression dans l'ordre original...");
                        return graph.Keys.ToList();
                    }
                }
            }

            return result;
        }

        static void DeleteEntitiesInOrder(IOrganizationService service, List<string> entityNames)
        {
            Console.WriteLine("\n" + new string('=', 60));
            Console.WriteLine("DÉBUT DE LA SUPPRESSION");
            Console.WriteLine(new string('=', 60) + "\n");

            int successCount = 0;
            int failureCount = 0;
            List<string> errors = new List<string>();
            List<string> successList = new List<string>();

            foreach (string entityName in entityNames)
            {
                try
                {
                    int currentIndex = entityNames.IndexOf(entityName) + 1;
                    Console.WriteLine($"[{currentIndex}/{entityNames.Count}] Suppression de '{entityName}'");

                    // ÉTAPE 1: Récupérer et supprimer toutes les dépendances
                    Console.WriteLine("  ? Analyse des dépendances CRM...");
                    bool dependenciesRemoved = RemoveAllDependencies(service, entityName);

                    if (!dependenciesRemoved)
                    {
                        Console.WriteLine("  ? Des dépendances persistent, tentative de suppression forcée...");
                    }

                    // ÉTAPE 2: Supprimer les vues système
                    Console.Write("  ? Suppression des vues système... ");
                    DeleteSystemViews(service, entityName);
                    Console.WriteLine("?");

                    // ÉTAPE 3: Supprimer les formulaires
                    Console.Write("  ? Suppression des formulaires... ");
                    DeleteForms(service, entityName);
                    Console.WriteLine("?");

                    // ÉTAPE 4: Supprimer les graphiques
                    Console.Write("  ? Suppression des graphiques... ");
                    DeleteCharts(service, entityName);
                    Console.WriteLine("?");

                    // ÉTAPE 5: Supprimer les dashboards
                    Console.Write("  ? Suppression des dashboards... ");
                    DeleteDashboards(service, entityName);
                    Console.WriteLine("?");

                    // ÉTAPE 6: Supprimer l'entité
                    Console.Write("  ? Suppression de l'entité... ");
                    var request = new DeleteEntityRequest
                    {
                        LogicalName = entityName
                    };
                    service.Execute(request);
                    Console.WriteLine("?");

                    Console.WriteLine($"  ? '{entityName}' SUPPRIMÉE AVEC SUCCÈS\n");
                    successCount++;
                    successList.Add(entityName);

                    System.Threading.Thread.Sleep(2000);
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"  ? ERREUR lors de la suppression");
                    failureCount++;
                    string errorMsg = ex.InnerException?.Message ?? ex.Message;
                    errors.Add($"{entityName}: {errorMsg}");
                    Console.WriteLine($"  Détails: {errorMsg}\n");
                }
            }

            // Résumé final
            Console.WriteLine(new string('=', 60));
            Console.WriteLine("RÉSUMÉ DE LA SUPPRESSION");
            Console.WriteLine(new string('=', 60));
            Console.WriteLine($"Total d'entités traitées : {entityNames.Count}");
            Console.WriteLine($"? Réussies               : {successCount}");
            Console.WriteLine($"? Échouées               : {failureCount}");

            if (successList.Count > 0)
            {
                Console.WriteLine("\n? Entités supprimées avec succès:");
                foreach (var entity in successList)
                {
                    Console.WriteLine($"  • {entity}");
                }
            }

            if (errors.Count > 0)
            {
                Console.WriteLine("\n? Erreurs détaillées:");
                foreach (var error in errors)
                {
                    Console.WriteLine($"  • {error}");
                }

                string errorFile = $"errors_{DateTime.Now:yyyyMMdd_HHmmss}.txt";
                File.WriteAllLines(errorFile, errors);
                Console.WriteLine($"\n? Erreurs sauvegardées dans: {errorFile}");
            }

            Console.WriteLine(new string('=', 60));
        }

        static bool RemoveAllDependencies(IOrganizationService service, string entityName)
        {
            try
            {
                // Récupérer l'ID de l'entité
                var retrieveEntityRequest = new RetrieveEntityRequest
                {
                    LogicalName = entityName,
                    EntityFilters = EntityFilters.Entity
                };
                var retrieveEntityResponse = (RetrieveEntityResponse)service.Execute(retrieveEntityRequest);
                var entityId = retrieveEntityResponse.EntityMetadata.MetadataId;

                if (!entityId.HasValue)
                {
                    Console.WriteLine("    ? Impossible de récupérer l'ID de l'entité");
                    return false;
                }

                // Récupérer toutes les dépendances
                var dependencyRequest = new RetrieveDependenciesForDeleteRequest
                {
                    ComponentType = 1, // Entity
                    ObjectId = entityId.Value
                };

                RetrieveDependenciesForDeleteResponse dependencyResponse;
                try
                {
                    dependencyResponse = (RetrieveDependenciesForDeleteResponse)service.Execute(dependencyRequest);
                }
                catch
                {
                    Console.WriteLine("    ? Aucune dépendance trouvée");
                    return true;
                }

                var dependencies = dependencyResponse.EntityCollection.Entities;

                if (dependencies.Count == 0)
                {
                    Console.WriteLine("    ? Aucune dépendance trouvée");
                    return true;
                }

                Console.WriteLine($"    ? {dependencies.Count} dépendance(s) trouvée(s)");

                // Supprimer chaque dépendance
                int removedCount = 0;
                foreach (var dependency in dependencies)
                {
                    try
                    {
                        int componentType = dependency.GetAttributeValue<int>("dependentcomponenttype");
                        Guid componentId = dependency.GetAttributeValue<Guid>("dependentcomponentobjectid");

                        string componentName = GetComponentTypeName(componentType);
                        Console.WriteLine($"      - Suppression: {componentName} ({componentId})");

                        bool removed = RemoveDependentComponent(service, componentType, componentId, entityName);
                        if (removed)
                        {
                            removedCount++;
                            Console.WriteLine($"        ? Supprimé");
                        }
                        else
                        {
                            Console.WriteLine($"        ? Non supprimé (peut être géré automatiquement)");
                        }
                    }
                    catch (Exception ex)
                    {
                        Console.WriteLine($"        ? Erreur: {ex.Message}");
                    }
                }

                Console.WriteLine($"    ? {removedCount}/{dependencies.Count} dépendances supprimées");
                return removedCount == dependencies.Count;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"    ? Erreur lors de la suppression des dépendances: {ex.Message}");
                return false;
            }
        }

        static bool RemoveDependentComponent(IOrganizationService service, int componentType, Guid componentId, string entityName)
        {
            try
            {
                switch (componentType)
                {
                    case 26: // View (SavedQuery)
                        service.Delete("savedquery", componentId);
                        return true;

                    case 60: // SystemForm
                        service.Delete("systemform", componentId);
                        return true;

                    case 59: // Chart (SavedQueryVisualization)
                        service.Delete("savedqueryvisualization", componentId);
                        return true;

                    case 62: // Workflow/Process
                        service.Delete("workflow", componentId);
                        return true;

                    case 29: // Business Process Flow
                        service.Delete("workflow", componentId);
                        return true;

                    case 65: // Hierarchy Rule
                        service.Delete("hierarchyrule", componentId);
                        return true;

                    case 18: // Entity Relationship
                        // Les relations sont gérées automatiquement lors de la suppression de l'entité
                        return false;

                    case 9: // Attribute
                        // Les attributs sont gérés automatiquement lors de la suppression de l'entité
                        return false;

                    case 10: // Entity Key
                        try
                        {
                            var deleteKeyRequest = new DeleteEntityKeyRequest
                            {
                                EntityLogicalName = entityName,
                                Name = componentId.ToString()
                            };
                            service.Execute(deleteKeyRequest);
                            return true;
                        }
                        catch
                        {
                            return false;
                        }

                    case 25: // Relationship (1:N or N:1)
                    case 63: // Relationship (N:N)
                        // Les relations sont gérées automatiquement
                        return false;

                    default:
                        Console.WriteLine($"          Type de composant non géré: {componentType}");
                        return false;
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"          Erreur de suppression: {ex.Message}");
                return false;
            }
        }

        static string GetComponentTypeName(int componentType)
        {
            switch (componentType)
            {
                case 1: return "Entity";
                case 2: return "Attribute";
                case 9: return "Option Set";
                case 10: return "Entity Relationship";
                case 18: return "Entity Key";
                case 20: return "Role";
                case 25: return "Relationship (1:N)";
                case 26: return "View";
                case 29: return "Business Process Flow";
                case 59: return "Chart";
                case 60: return "Form";
                case 61: return "Web Resource";
                case 62: return "Workflow";
                case 63: return "Relationship (N:N)";
                case 65: return "Hierarchy Rule";
                case 70: return "Field Security Profile";
                case 90: return "Plugin Type";
                case 91: return "Plugin Assembly";
                case 92: return "SDK Message Processing Step";
                case 93: return "SDK Message Processing Step Image";
                default: return $"Unknown ({componentType})";
            }
        }

        static void DeleteSystemViews(IOrganizationService service, string entityName)
        {
            try
            {
                var query = new QueryExpression("savedquery")
                {
                    ColumnSet = new ColumnSet("savedqueryid", "name"),
                    Criteria = new FilterExpression
                    {
                        Conditions =
                        {
                            new ConditionExpression("returnedtypecode", ConditionOperator.Equal, entityName)
                        }
                    }
                };

                var views = service.RetrieveMultiple(query);
                foreach (var view in views.Entities)
                {
                    try
                    {
                        service.Delete("savedquery", view.Id);
                    }
                    catch { }
                }
            }
            catch { }
        }

        static OrganizationServiceProxy GetOrganizationService(ConnectionConfig config)
        {
            try
            {
                Uri orgUri = new Uri(config.ServerUrl + "/XRMServices/2011/Organization.svc");

                ClientCredentials credentials = new ClientCredentials();

                if (!string.IsNullOrEmpty(config.Domain))
                {
                    credentials.Windows.ClientCredential = new System.Net.NetworkCredential(
                        config.Username,
                        config.Password,
                        config.Domain
                    );
                }
                else
                {
                    credentials.UserName.UserName = config.Username;
                    credentials.UserName.Password = config.Password;
                }

                Console.WriteLine($"Connexion à: {orgUri}");
                var serviceProxy = new OrganizationServiceProxy(orgUri, null, credentials, null);
                serviceProxy.EnableProxyTypes();

                serviceProxy.Execute(new WhoAmIRequest());

                return serviceProxy;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Erreur de connexion: {ex.Message}");
                return null;
            }
        }

        static List<string> GetEntitiesToDelete()
        {
            List<string> entities = new List<string>();

            string inputFile = @"C:\Users\crmadmin\source\repos\RemoveEntity\RemoveEntity\entities_to_delete.txt";
            if (File.Exists(inputFile))
            {
                Console.WriteLine($"\nLecture du fichier: {inputFile}");
                entities = File.ReadAllLines(inputFile)
                    .Where(line => !string.IsNullOrWhiteSpace(line) && !line.Trim().StartsWith("#"))
                    .Select(line => line.Trim().ToLower())
                    .Distinct()
                    .ToList();

                Console.WriteLine($"? {entities.Count} entité(s) trouvée(s)");
                return entities;
            }

            Console.WriteLine("\nEntrez les noms logiques des entités à supprimer (une par ligne).");
            Console.WriteLine("Exemples: etel_test, etel_account, etel_access");
            Console.WriteLine("Tapez 'fin' pour terminer la saisie.\n");

            while (true)
            {
                Console.Write("Entité: ");
                string input = Console.ReadLine()?.Trim().ToLower();

                if (string.IsNullOrEmpty(input) || input == "fin")
                    break;

                if (!entities.Contains(input))
                {
                    entities.Add(input);
                    Console.WriteLine($"? Ajouté: {input}");
                }
                else
                {
                    Console.WriteLine($"? Déjà dans la liste: {input}");
                }
            }

            if (entities.Count > 0)
            {
                File.WriteAllLines(inputFile, entities);
                Console.WriteLine($"\n? Liste sauvegardée dans {inputFile}");
            }

            return entities;
        }
